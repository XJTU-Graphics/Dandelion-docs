275template<typename DataType>
276constexpr GLenum get_GL_type_enum()
277{
278 if constexpr (std::is_same_v<DataType, char>) {
279 return GL_BYTE;
280 } else if constexpr (std::is_same_v<DataType, unsigned char>) {
281 return GL_UNSIGNED_BYTE;
282 } else if constexpr (std::is_same_v<DataType, int>) {
283 return GL_INT;
284 } else if constexpr (std::is_same_v<DataType, unsigned int>) {
285 return GL_UNSIGNED_INT;
286 } else if constexpr (std::is_same_v<DataType, float>) {
287 return GL_FLOAT;
288 } else if constexpr (std::is_same_v<DataType, double>) {
289 return GL_DOUBLE;
290 } else {
291 return GL_NONE;
292 }
293}

336void ArrayBuffer<T, size>::update(size_t index, const Eigen::Vector3f& value)
337{
338 const GLintptr offset = index * 3 * sizeof(float);
339 data[index * 3] = value.x();
340 data[index * 3 + 1] = value.y();
341 data[index * 3 + 2] = value.z();
342 bind();
343 glBufferSubData(GL_ARRAY_BUFFER, offset, 3 * sizeof(float), value.data());
344}
